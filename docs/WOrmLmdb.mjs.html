import { open, ABORT, keyValueToBuffer } from 'lmdb'
// import mingo from 'mingo' //mingo內未更新import寫法, 會導致ERR_UNSUPPORTED_DIR_IMPORT, 故須改用require引入使用

/**
 * 操作資料庫(LMDB)
 *
 * 回傳物件為EventEmitter，除各操作函數外另發出change與error兩事件，供呼叫端於單一處集中觀察資料異動與失敗。
 * 事件僅為附加通知，其所送出之資訊皆另有正規管道(操作結果經resolve、整批性錯誤經reject、逐筆失敗經該筆之err欄位)，
 * 故不監聽亦能取得完整資訊，且監聽與否不改變任何操作之回傳值。
 *
 * change事件，參數為(mode, data, res)，於資料實際異動成功後發出：
 * mode為操作別字串，可為'insert'、'insertBulk'、'save'、'del'、'delAll'；save內若逐筆走自動插入則該筆另發出mode為'insert'之事件，於整批定案後、整批事件之前發出。
 * data為本次操作之輸入數據，delAll固定為null。
 * res為本次操作之回傳結果之複本，改寫之不影響回傳結果。
 * 逐筆函數以整批為單位發出一次而不逐筆發出，select與selectByPk不發出本事件。
 *
 * error事件，參數為(mode, data, err)，於操作發生錯誤時發出：
 * mode為操作別字串，可為'select'、'selectByPk'、'insert'、'insertBulk'、'save'、'del'、'delAll'。
 * data為本次操作之輸入數據，無輸入數據者為null。
 * err為錯誤訊息字串，內容與正規管道所送出者一致。
 * 整批性錯誤於reject之前發出；逐筆失敗於整批定案後依序發出，每筆一次，整批因整批性錯誤而reject者不發出。
 * 註: 逐筆失敗時整批仍resolve，故收到本事件不表示該次呼叫失敗；正常結果(如查無數據、主鍵未命中、全數已存在)不發出本事件。
 *
 * 讀取(select、selectByPk，以及save、del、delAll內之讀取)可見本次呼叫前已提交之資料，含同一程序之其他實例與其他程序之提交；惟select於useCache開啟且快取命中時回傳快取。
 *
 * 主鍵(id)經LMDB鍵編碼後不可超過1978位元組(UTF-8；首字元碼小於28者另計1位元組跳脫)：insert與insertBulk遇之整批reject而不寫入任何一筆，save該筆ok為0；
 * 讀取側(selectByPk、del與save內之讀取)對超過者一律視為未命中。
 * 註: 值非物件之紀錄(他方寫入)於insert與insertBulk視為已存在(以鍵判定)，於selectByPk、save與del視為不存在。
 *
 * @class
 * @param {Object} [opt={}] 輸入設定物件，預設{}
 * @param {String} [opt.url='_db'] 輸入資料庫用資料夾字串，預設'_db'
 * @param {String} [opt.db='worm'] 輸入使用資料庫名稱字串，預設'worm'
 * @param {String} [opt.cl='test'] 輸入使用資料表名稱字串，預設'test'
 * @param {Boolean} [opt.useCache=false] 輸入是否使用select快取，快取只反映本實例之寫入，不反映同一程序之其他實例或其他程序之寫入，故僅適用於同一資料表只由本實例存取之情形；delAll一律直讀資料庫而不經快取，預設false
 * @param {Boolean} [opt.autoGenPk=true] 輸入insert與save於數據未帶有效主鍵(本套件為id欄位)時是否自動產生主鍵值，預設true；給false代表主鍵改由呼叫端自備，套件不產生亦不檢查其唯一性與格式，未帶有效主鍵者將reject
 * @returns {Object} 回傳操作資料庫物件，各事件功能詳見說明
 */
function WOrmLmdb(opt = {}) {

    //_cache
    let _cache = null

    //url
    let url = get(opt, 'url')
    if (!isestr(url)) {
        url = './_db'
    }

    //db
    let db = get(opt, 'db')
    if (!isestr(db)) {
        db = 'worm'
    }

    //cl
    let cl = get(opt, 'cl')
    if (!isestr(cl)) {
        cl = 'test'
    }

    //useCache
    let useCache = get(opt, 'useCache')
    if (!isbol(useCache)) {
        useCache = false
    }

    //autoGenPk, 主鍵由誰產生為整個資料表之政策, 故為建構層設定, 不可於insert與save之option逐次覆寫,
    //否則同一資料表將混入兩種來源之主鍵而難以追溯
    let autoGenPk = get(opt, 'autoGenPk')
    if (!isbol(autoGenPk)) {
        autoGenPk = true
    }

    //storage
    let storage = `${url}/${db}/${cl}`
    // console.log('storage',storage)

    // client
    let client = open({
        path: storage,
        compression: true,
        useVersions: true,
    })

    //ee
    let ee = evem()

    //readAll, 由資料庫讀取全部數據而不經快取, mode為呼叫之操作別, 供已關閉時之錯誤訊息使用, withKey為true時各筆改回{key,value}
    let readAll = (mode, withKey = false) => {

        //prepareRead, 須與其後之讀取同步執行
        prepareRead(mode)

        //ltdt, 須同步迭代而不可用for await: for await逐筆讓出, 併發之select會交錯持有讀交易,
        //讀前重設遇交易使用中且已有新提交時lmdb-js即另開讀交易, 致reader slot(預設126, 跨程序共用)耗盡而MDB_READERS_FULL
        //須於finally呼叫return()歸還cursor: next()內拋錯(如數據無法解碼)時for...of不會呼叫return(),
        //讀交易將被孤立而持續占用reader slot直到GC, 反覆失敗即耗盡全環境(含其他程序)之reader slot
        let ltdt = []
        let it = client.getRange()[Symbol.iterator]()
        try {
            for (let r = it.next(); !r.done; r = it.next()) {
                ltdt.push(withKey ? { key: r.value.key, value: r.value.value } : r.value.value)
            }
        }
        finally {
            try {
                it.return()
            }
            catch (err) {
                //歸還cursor之錯誤不可掩蓋讀取之結果或原錯誤
            }
        }

        return ltdt
    }

    //hasProtoName, 檢核後複本之欄名是否同於Object.prototype之成員(constructor、toString、valueOf、hasOwnProperty等):
    //此類欄名屬白名單內之合法欄名, 但mingo以屬性存取取值, 數據無此自有欄位時會取到繼承之成員(函數)而非缺欄,
    //使{constructor:null}無命中、{constructor:{$ne:null}}命中全部(delAll因而刪光); 欄名只會出現於複本之鍵(含$and/$or內之子條件)
    let hasProtoName = (fc) => {
        for (let k of Object.keys(fc)) {
            if (k === '$and' || k === '$or') {
                for (let sub of fc[k]) {
                    if (hasProtoName(sub)) {
                        return true
                    }
                }
            }
            else if (k in Object.prototype) {
                return true
            }
        }
        return false
    }

    //mkMatch, 依檢核後複本建立逐筆命中判定, select與delAll共用同一判定(系列最低要求①: select命中之數據即delAll刪除之數據);
    //欄名同於Object.prototype成員者, 改以null原型之淺複本(只含自有可列舉欄位)判定, 令其與一般不存在之欄位同為缺欄語意, 命中者仍回傳原數據
    let mkMatch = (fc) => {
        let q = new mingo.Query(fc)
        if (!hasProtoName(fc)) {
            return (v) => q.test(v)
        }
        return (v) => q.test(isobj(v) ? Object.assign(Object.create(null), v) : v)
    }

    //getData, 供select使用, 依useCache使用快取, mode為呼叫之操作別
    let getData = async(mode) => {

        //check
        if (useCache && isarr(_cache)) {
            return cloneDeep(_cache) //與外部使用數據脫勾
        }

        //ltdt
        let ltdt = readAll(mode)

        if (useCache) {
            _cache = ltdt
            return cloneDeep(ltdt) //與外部使用數據脫勾
        }
        return ltdt
    }

    //isClosed, close()後lmdb之status由'open'轉'closing', 待未完成之寫入結束後再轉'closed', 二者皆為終態,
    //一律視為已關閉; status僅此三值, 且'open'於open()回傳前即已設定, 故無開啟中而須等待之狀態
    let isClosed = () => {
        return client.status !== 'open'
    }

    //getClosedMsg, 已關閉之錯誤訊息
    let getClosedMsg = (mode) => {
        return `can not ${mode} by closed client, instance can not be used after close()`
    }

    //procClosed, 各函數入口之終態守門: close()後(含關閉進行中)再操作屬整批性錯誤, 立即reject令誤用當場可見
    //不可靜默回正常空值(如null或未命中), 否則[已關閉]與[查無資料]同形,
    //去重類呼叫端將把每筆判為未存在而fail-open, 整批重送下游昂貴動作
    let procClosed = (mode, data) => {
        let msg = getClosedMsg(mode)
        emitError(mode, data, msg)
        return Promise.reject(new Error(msg))
    }

    //prepareRead, 寫交易外之讀取前呼叫, 須與其後之讀取同步執行(中間不可await), 令close()無從插入:
    //1.已關閉則拋錯: 關閉進行中呼叫lmdb之讀取, lmdb-js會於計時器內拋出未捕捉例外而使程序崩潰,
    //  入口守門擋不住入口通過後才close()者(save、del之逐筆讀取於其後之microtask才執行), 故須於讀取處再擋
    //2.重設讀交易: lmdb-js之讀交易屬於每次open所得之實例, 延至下一個timers階段才重設; 本實例之寫入於提交時即重設,
    //  其他程序與同一程序之其他實例(另一次open)之提交則否, 於此前不可見; 讀前重設令讀取可見本次呼叫前已提交之資料,
    //  寫交易內之讀取已取最新資料, 不需重設
    let prepareRead = (mode) => {
        if (isClosed()) {
            throw new Error(getClosedMsg(mode))
        }
        client.resetReadTxn()
    }

    //emitChange, 資料實際異動成功後發出, 事件僅為附加通知不承擔回傳義務
    //一律包try/catch, 令訂閱函數自身拋錯不影響本次操作之結果
    //有訂閱者時傳入res之複本: res即回傳予呼叫端之物件, 訂閱者若改寫之將改變回傳結果(T10.1第4條)
    let emitChange = (mode, data, res) => {
        try {
            if (ee.listenerCount('change') > 0) {
                ee.emit('change', mode, data, cloneDeep(res))
            }
        }
        catch (err) {
            console.log(err)
        }
    }

    //emitError, 操作發生錯誤時發出, 錯誤訊息一律轉為字串
    //此try/catch除攔訂閱函數之例外外另有必要: Node之EventEmitter於'error'無監聽者時會將錯誤直接拋出,
    //若不攔則同一次操作會因呼叫端有無註冊監聽而走向不同結果
    let emitError = (mode, data, err) => {
        try {
            ee.emit('error', mode, data, getErrMsg(err))
        }
        catch (errEmit) {
            console.log(errEmit)
        }
    }

    //getErrMsg, 取錯誤訊息字串, 供逐筆結果之err欄位與error事件使用
    let getErrMsg = (err) => {
        let m = get(err, 'message')
        if (isestr(m)) {
            return m
        }
        return String(err)
    }

    //procPk, 依autoGenPk處理各數據之主鍵, 供insert、insertBulk與save共用
    //各筆須為物件: 陣列、Date、Map、Set等經編碼後不保留id欄位, 寫入後無從以主鍵管理, 屬呼叫端未履行契約之整批性錯誤;
    //autoGenPk為true時補值; 為false時不補值, 未帶有效主鍵者屬呼叫端未履行契約, 拋出為整批性錯誤,
    //且檢查於任何寫入之前完成, 故不會有部份筆數已寫入而整批失敗之情形
    let procPk = (data, fnName) => {
        each(data, function(v, k) {
            if (!isobj(v)) {
                throw new Error(`can not ${fnName} by data[${k}] which is not an object`)
            }
        })
        if (autoGenPk) {
            return map(data, function(v) {
                if (!isestr(v.id)) {
                    v.id = genIDSeq()
                }
                return v
            })
        }
        each(data, function(v, k) {
            if (!isestr(get(v, 'id'))) {
                throw new Error(`can not ${fnName} by data[${k}] without valid id when autoGenPk is false`)
            }
        })
        return data
    }

    //maxKeySize, lmdb之鍵經編碼後之位元組數上限(預設1978)
    let maxKeySize = get(client, 'maxKeySize', 1978)

    //keyFits, 鍵經lmdb之鍵編碼後之位元組數是否不超過maxKeySize, 寫入側與讀取側共用同一判定
    //以lmdb公開匯出之keyValueToBuffer量測, 與lmdb之判定一致(首字元碼小於28者多1位元組跳脫, 中文每字3位元組);
    //其暫存緩衝區為8192位元組, 超過者拋錯, 必超過maxKeySize
    let keyFits = (key) => {
        let n = -1
        try {
            n = keyValueToBuffer(key).length
        }
        catch (err) {
            n = -1
        }
        return n >= 0 && n <= maxKeySize
    }

    //procWritable, 供insert與insertBulk於任何寫入之前檢查各筆可寫入, 任一筆不可寫入即拋出為整批性錯誤而不寫入任何一筆
    //1.主鍵經lmdb之鍵編碼後之位元組數不可超過maxKeySize: 超過者lmdb於寫入呼叫內同步拋錯, insert一次排入多筆條件寫入,
    //  拋錯前已排入者仍於其後提交(部份寫入且reject當下不可見); 壞鍵為該事件迴圈之第一筆寫入時, lmdb-js之close()不等候其已排程之批次,
    //  該批次於關閉後啟動寫入而拋出未捕捉例外, 行程崩潰
    //2.數據須可被編碼(如Symbol、超出64位元之BigInt、循環參照不可編碼): 編碼同樣於寫入呼叫內同步執行, 理由同上
    //訊息以lmdb或編碼器之原訊息為後綴, 與lmdb原本拋出者一致而可辨識
    let procWritable = (data, fnName) => {
        each(data, function(v, k) {
            if (!keyFits(v.id)) {
                throw new Error(`can not ${fnName} by data[${k}], Key size is larger than the maximum key size (${maxKeySize})`)
            }
            try {
                client.encoder.encode(v)
            }
            catch (err) {
                throw new Error(`can not ${fnName} by data[${k}], ${getErrMsg(err)}`)
            }
        })
    }

    //getValue, mode為呼叫之操作別, 供已關閉時之錯誤訊息使用
    //讀取之錯誤(如已關閉、數據無法解碼)不可攔下回null, 否則與[查無資料]同形, 須交由呼叫端之錯誤處理;
    //lmdb-js於查無數據時回undefined而不拋錯, 故會拋出者皆為真錯誤
    let getValue = async (key, mode) => {
        prepareRead(mode)

        //鍵超過lmdb之鍵長上限者必不存在, 視為未命中而不呼叫lmdb之讀取: 其讀取API以4096位元組之緩衝寫鍵,
        //4093位元組起拋錯, 否則同為不存在之鍵將依長度而回未命中或錯誤; 須於prepareRead之後, 令已關閉之判斷優先
        if (!keyFits(key)) {
            return undefined
        }

        return client.get(key)
    }

    /**
     * 查詢數據
     *
     * 條件find僅支援系列白名單(規則同delAll之第1-3點，見delAll)，與delAll共用同一檢核，select(find)命中之數據即delAll(find)刪除之數據
     *
     * @memberOf WOrmLmdb
     * @param {Object} [find={}] 輸入查詢條件物件(系列白名單，規則同delAll之第1-3點); 未給、undefined或{}回傳全部數據; 非條件物件或白名單外運算子整批reject
     * @returns {Promise} 回傳Promise，resolve回傳數據，reject回傳錯誤訊息
     */
    async function select(find = {}) {

        //check client, 已關閉(含關閉進行中)為終態, 於任何檢查與寫入之前先行攔下(T4: 實例已關閉屬整批性錯誤)
        if (isClosed()) {
            return procClosed('select', null)
        }

        //fc, 條件之入口檢核(R-FIND-01): 入口順序為closed→find檢核→讀取, 不合者依T10.3先發出error事件再reject(訊息原樣, 不經下方讀取階段之錯誤處理),
        //其後一律只用檢核所回傳之全新複本(null原型, 與呼叫端之find脫勾), 不再讀取呼叫端之find
        let fc = null
        try {
            fc = cvOrmFind(find)
        }
        catch (err) {
            emitError('select', null, err)
            return Promise.reject(err)
        }

        let isErr = false
        let res = null

        try {

            //ltdt
            let ltdt = await getData('select')
            // console.log('select ltdt',ltdt)

            //filter, 複本無自有鍵(未給、undefined、{})為全部
            if (Reflect.ownKeys(fc).length > 0) {

                //find, 與delAll共用同一逐筆判定
                res = ltdt.filter(mkMatch(fc))
                // console.log('select(find) ltdt',res)

            }
            else {
                res = ltdt
            }

            //check
            if (!isarr(res)) {
                isErr = true
                res = `can not select by find[${JSON.stringify(fc)}]`
            }

        }
        catch (err) {
            isErr = true
            res = err
        }
        // console.log('res', res)

        //check
        if (isErr) {

            //emit, 整批性錯誤須於reject之前發出
            emitError('select', null, res)

            return Promise.reject(res)
        }

        return res
    }

    /**
     * 由主鍵查詢單筆數據，因直接由LMDB取值，不需如select提取全表數據再過濾，故數據量大時效能較佳
     * 註: 本套件之主鍵欄位固定為id，尚未支援指定其他欄位為主鍵
     *
     * @memberOf WOrmLmdb
     * @param {String} pk 輸入主鍵值字串，本套件之主鍵欄位為id
     * @returns {Promise} 回傳Promise，resolve回傳數據物件，若無此主鍵則回傳null(主鍵經LMDB鍵編碼後超過1978位元組者必不存在，亦回傳null)，reject回傳錯誤訊息
     */
    async function selectByPk(pk) {

        //check client, 已關閉(含關閉進行中)為終態, 於任何檢查與寫入之前先行攔下(T4: 實例已關閉屬整批性錯誤)
        if (isClosed()) {
            return procClosed('selectByPk', null)
        }
        let isErr = false
        let res = null

        try {

            //check
            if (!isestr(pk)) {
                //未給有效主鍵值視為查無數據
                return null
            }

            //查找資料表內pk, 讀取之錯誤(含已關閉)由下方catch轉為reject
            let v = await getValue(pk, 'selectByPk')

            //check, 判定基準與insert、save、del內對既有數據之認定一致
            if (iseobj(v)) {
                res = v
            }
            else {
                //不存在此主鍵, 回傳null
                res = null
            }

        }
        catch (err) {
            isErr = true
            res = err
        }

        //check
        if (isErr) {

            //emit, 整批性錯誤須於reject之前發出
            emitError('selectByPk', null, res)

            return Promise.reject(res)
        }

        return res
    }

    /**
     * 插入數據
     *
     * @memberOf WOrmLmdb
     * @param {Object|Array} data 輸入數據物件或陣列
     * @param {Object} [option={}] 輸入設定物件，預設為{}
     * @param {Boolean} [option.returnList=false] 輸入是否改回傳逐筆結果布林值，預設false。給true時回傳與輸入等長保序之陣列，各筆為{n,nInserted,ok}，nInserted為1即該筆為新增；聚合計數只回答有幾筆是新的，逐筆結果方能回答是哪幾筆，供下游僅對新資料執行昂貴動作。回傳形式由呼叫點靜態決定，共用之結果處理程式碼不可跨不同取值之呼叫點混用
     * @returns {Promise} 回傳Promise，resolve依returnList回傳插入結果：預設回傳聚合物件{n,nInserted,ok}，n為輸入筆數、nInserted為實際插入筆數；returnList為true時回傳逐筆陣列，各筆n恆為1(主鍵命中或經插入)、ok恆為1(insert之錯誤皆屬整批性而reject，不進逐筆)；reject回傳錯誤訊息。整批性錯誤(實例已關閉、autoGenPk為false而未帶有效主鍵、數據元素非物件、主鍵經LMDB鍵編碼後超過1978位元組、數據無法序列化)皆於任何寫入之前檢出而reject，不寫入任何一筆
     */
    async function insert(data, option = {}) {

        //check client, 已關閉(含關閉進行中)為終態, 於任何檢查與寫入之前先行攔下(T4: 實例已關閉屬整批性錯誤)
        if (isClosed()) {
            return procClosed('insert', data)
        }
        let isErr = false

        //returnList
        let returnList = get(option, 'returnList')
        if (!isbol(returnList)) {
            returnList = false
        }

        //check
        if (!iseobj(data) && !isearr(data)) {
            if (returnList) {
                //輸入無效, 對齊save與del之T5規定回傳空陣列
                return []
            }
            return {
                n: 0,
                nInserted: 0,
                ok: 1,
            }
        }

        //res
        let res = null
        try {

            //cloneDeep, 與外部數據脫勾; 置於try內, 無法複製之輸入(如過深之巢狀、取值拋錯之getter)同屬整批性錯誤而先發出error事件
            data = cloneDeep(data)

            //check
            if (!isarr(data)) {
                data = [data]
            }

            //check id
            data = procPk(data, 'insert')

            //check writable, 須於任何寫入之前完成, 其後之寫入呼叫不再同步拋錯, 故不會部份寫入
            procWritable(data, 'insert')

            //each
            //ifNoExists, 由LMDB於同一寫交易內原子完成[檢查v.id未存在]與[寫入], 併發時同一v.id僅有一次成功,
            //同批含重複v.id時亦僅首筆成功, 故不逐筆await而一次送出, 由LMDB寫執行緒依序處理並合併為批次寫交易
            //注意: 回調函數須為同步函數, 且client.put須於回調函數內直接呼叫,
            //若將client.put置於await之後, 該寫入會逸出條件批次而變成無條件寫入, 並使回傳值與實際狀態不一致
            let nAll = size(data)
            let ltdone = await Promise.all(map(data, function(v) {
                // console.log(v)
                return client.ifNoExists(v.id, () => {
                    client.put(v.id, v)
                })
            }))

            //nPush
            let nPush = 0
            each(ltdone, function(done) {
                if (done) {
                    //未存在v.id, 已寫入
                    nPush++
                }
                else {
                    //已存在v.id則不push
                }
            })

            //res
            if (returnList) {
                //逐筆結果, 與輸入等長保序, 元素沿用逐筆結果之家族形狀(同save與del之元素)
                //n恆為1(主鍵命中既有或經插入而產生), ok恆為1(insert之錯誤皆屬整批性而reject, 不進逐筆),
                //資訊由nInserted承載, filter(v=>v.nInserted===1)之長度必等於聚合模式之nInserted
                res = map(ltdone, function(done) {
                    return {
                        n: 1,
                        nInserted: done ? 1 : 0,
                        ok: 1,
                    }
                })
            }
            else {
                res = {
                    n: nAll,
                    nInserted: nPush,
                    ok: 1,
                }
            }

        }
        catch (err) {
            isErr = true
            res = err
        }

        //update, 不能保證插入多少, 一律重設快取
        _cache = null

        //emit, 於change可能須使用select, 故須放在重設快取之後
        if (!isErr) {
            emitChange('insert', data, res)
        }

        //check
        if (isErr) {

            //emit, 整批性錯誤須於reject之前發出
            emitError('insert', data, res)

            return Promise.reject(res)
        }

        return res
    }

    /**
     * 批次插入數據，全批視為一個單位，全部插入成功或一筆都不寫入；主鍵衝突以外之整批性錯誤(同insert)亦於任何寫入之前檢出而reject
     * 註: 本函數非insert之加速版，兩者衝突政策不同。insert於主鍵已存在時跳過該筆而整批ok為1，
     * 本函數則整批reject且不寫入任何一筆；同批含重複主鍵者亦視為衝突。確無衝突時兩者結果相同
     * 註: 於本套件不會較insert快，因insert本即以Promise.all一次送出全部條件寫入而非逐筆await，
     * 提供本函數係為與其他w-orm系列套件維持同一組函數，令呼叫端得於各套件間替換而不須改寫呼叫
     *
     * @memberOf WOrmLmdb
     * @param {Object|Array} data 輸入數據物件或陣列
     * @returns {Promise} 回傳Promise，resolve回傳插入結果物件{n,nInserted,ok}，n為輸入筆數、nInserted成功時恆等於n，任一筆主鍵已存在則reject回傳錯誤訊息
     */
    async function insertBulk(data) {

        //check client, 已關閉(含關閉進行中)為終態, 於任何檢查與寫入之前先行攔下(T4: 實例已關閉屬整批性錯誤)
        if (isClosed()) {
            return procClosed('insertBulk', data)
        }
        let isErr = false

        //check
        if (!iseobj(data) && !isearr(data)) {
            return {
                n: 0,
                nInserted: 0,
                ok: 1,
            }
        }

        //res
        let res = null

        //pkConflict, 置於try外供catch分辨衝突與已關閉
        let pkConflict = null

        try {

            //cloneDeep, 與外部數據脫勾; 置於try內, 無法複製之輸入同屬整批性錯誤而先發出error事件
            data = cloneDeep(data)

            //check
            if (!isarr(data)) {
                data = [data]
            }

            //check id
            data = procPk(data, 'insertBulk')

            //check writable, 交易回滾雖已保證全無寫入, 仍於寫入前檢出以與insert之訊息一致
            procWritable(data, 'insertBulk')

            //nAll
            let nAll = size(data)

            //childTransaction, 全有全無由交易回滾保證
            //註: 非同步之client.transaction於中止時並不回滾, 實測其內之put仍會落盤, 故此處須用childTransaction
            //註: 交易內之讀取讀得到同一交易稍早之put, 故同批含重複主鍵者亦會於此被偵測為衝突
            //註: 存在與否採鍵層判定(doesExist只判鍵而不解碼數據), 與insert之ifNoExists一致
            await client.childTransaction(() => {
                for (let v of data) {
                    if (client.doesExist(v.id)) {
                        pkConflict = v.id
                        return ABORT
                    }
                    client.put(v.id, v)
                }
            })

            //check, 衝突則整批視為失敗, 交易已回滾故無任何寫入
            if (isestr(pkConflict)) {
                throw new Error(`can not insertBulk by existed id[${pkConflict}]`)
            }

            //res, 未衝突則全數插入, 故nInserted恆等於n
            res = {
                n: nAll,
                nInserted: nAll,
                ok: 1,
            }

        }
        catch (err) {
            isErr = true
            res = err

            //入口通過後才close者, childTransaction之回調延後執行而遇關閉, lmdb回報Database is closed,
            //統一為已關閉之訊息(T4: 訊息明示closed); 主鍵衝突者交易已實際執行, 保留其衝突訊息
            if (isClosed() && !isestr(pkConflict)) {
                res = new Error(getClosedMsg('insertBulk'))
            }

        }

        //update, 不能保證插入多少, 一律重設快取
        _cache = null

        //emit, 於change可能須使用select, 故須放在重設快取之後
        if (!isErr) {
            emitChange('insertBulk', data, res)
        }

        //check
        if (isErr) {

            //emit, 整批性錯誤須於reject之前發出
            emitError('insertBulk', data, res)

            return Promise.reject(res)
        }

        return res
    }

    /**
     * 儲存數據
     *
     * @memberOf WOrmLmdb
     * @param {Object|Array} data 輸入數據物件或陣列
     * @param {Object} [option={}] 輸入設定物件，預設為{}
     * @param {boolean} [option.autoInsert=true] 輸入是否於儲存時發現原本無數據，則自動改以插入處理，預設為true
     * @returns {Promise} 回傳Promise，resolve回傳與輸入等長之儲存結果陣列，各筆為{n,nInserted,nModified,ok}，n為主鍵命中筆數(主鍵經LMDB鍵編碼後超過1978位元組者視為未命中)，單筆失敗者(如須插入而主鍵超過上述上限、數據無法序列化)ok為0並附err，reject回傳錯誤訊息；數據元素非物件等整批性錯誤於任何寫入之前reject；處理中途發生整批性錯誤(如操作進行中被close)時整批reject，先前已處理之筆不回滾且其逐筆事件不發出，呼叫端須重新讀取確認
     */
    async function save(data, option = {}) {

        //check client, 已關閉(含關閉進行中)為終態, 於任何檢查與寫入之前先行攔下(T4: 實例已關閉屬整批性錯誤)
        if (isClosed()) {
            return procClosed('save', data)
        }
        let isErr = false

        //check
        if (!iseobj(data) && !isearr(data)) {
            return []
        }

        //autoInsert
        let autoInsert = get(option, 'autoInsert', true)

        //res
        let res = null

        //evs, 逐筆之事件先緩存, 待整批定案(resolve)後依序發出; 整批性錯誤而reject者不發出(T10.1第1條、T10.2)
        let evs = []

        try {

            //cloneDeep, 與外部數據脫勾; 置於try內, 無法複製之輸入同屬整批性錯誤而先發出error事件
            data = cloneDeep(data)

            //check
            if (!isarr(data)) {
                data = [data]
            }

            //check id
            data = procPk(data, 'save')

            //pmSeries
            res = await pmSeries(data, async(v) => {

                //rest
                let rest = null

                //inserted, 供emit使用
                let inserted = false

                try {

                    //快速路徑, 先不開寫交易直接預讀, 內容相同者本就不須寫入, 可省去開啟寫交易之成本
                    //預讀值縱使已被其他寫入者更動而過期亦不影響正確性, 因內容相同時本次save等價於無操作,
                    //無操作可視為於預讀當下即已完成, 其後他人之寫入結果與[本次save先執行再輪到他人]相同
                    //前提為預讀所見不早於本次呼叫開始, 由getValue之讀前重設保證(含同一程序之其他實例與其他程序已提交者);
                    //若沿用呼叫前之快照, [預讀當下]將早於呼叫本身, 呼叫端於呼叫前已知之較新寫入會被略過而遺失
                    //註: 預讀僅用於判斷是否略過寫入, 不用於決定寫入內容, 寫入內容一律由交易內讀到之現值決定
                    let vp
                    try {
                        vp = await getValue(v.id, 'save')
                    }
                    catch (err) {
                        //已關閉者往外拋; 其餘錯誤放棄快速路徑而續走寫交易, 結果由交易內之讀取決定,
                        //快速路徑為省略寫交易之選擇性優化, 不可使原本可由寫交易完成之儲存失敗(如讀交易之reader slot不足)
                        if (isClosed()) {
                            throw err
                        }
                        vp = undefined
                    }
                    if (iseobj(vp) && isEqual(merge({}, vp, v), vp)) {
                        //合併後與現值相同不須寫入, 惟已命中v.id故n為1
                        return {
                            n: 1,
                            nInserted: 0,
                            nModified: 0,
                            ok: 1,
                        }
                    }

                    //existed, updated
                    let existed = false
                    let updated = false

                    //transaction, 由LMDB於同一寫交易內原子完成[查找v.id]與[合併寫入],
                    //併發時不會有兩方各自讀到同一舊值再各自寫入而導致遺失更新
                    //注意: 回調函數須為同步函數, client.get於寫交易內為同步取值, 不可使用await,
                    //若將client.get或client.put置於await之後, 該操作會逸出本交易而失去原子性
                    await client.transaction(() => {

                        //查找資料表內v.id, 鍵超過lmdb之鍵長上限者必不存在, 視為未命中而不呼叫lmdb之讀取(同getValue)
                        let vv = keyFits(v.id) ? client.get(v.id) : undefined

                        //existed
                        existed = iseobj(vv)

                        //check
                        if (existed) {
                            //已存在v.id

                            //vm, 合併後之結果
                            let vm = merge({}, vv, v)

                            //判定基準為[合併後結果與現值相同], 而非[待寫入物件與現值全等],
                            //令nModified忠實反映資料庫端是否真的寫入, 僅給部份欄位且值皆相同者不應回報已修改
                            if (isEqual(vm, vv)) {
                                //合併後與現值相同不須寫入
                            }
                            else {
                                //合併後與現值不同須更新

                                //put
                                client.put(v.id, vm)

                                updated = true
                            }
                        }
                        else {
                            //內容不存在, 若autoInsert則須插入
                            if (autoInsert) {

                                //put
                                client.put(v.id, v)

                                inserted = true
                            }
                        }

                    })

                    //rest, nInserted與nModified恆存在, 無對應行為者給0
                    if (inserted) {
                        rest = {
                            n: 1,
                            nInserted: 1,
                            nModified: 0,
                            ok: 1,
                        }
                    }
                    else if (updated) {
                        rest = {
                            n: 1,
                            nInserted: 0,
                            nModified: 1,
                            ok: 1,
                        }
                    }
                    else {
                        //未寫入, 有二:
                        // 已存在v.id而內容相同不更新, 已命中故n為1
                        // 不存在v.id且未開啟autoInsert, 未命中故n為0
                        //n為v.id之命中筆數, 與nModified[實際更新筆數]分屬不同語義, 不可混用
                        rest = {
                            n: existed ? 1 : 0,
                            nInserted: 0,
                            nModified: 0,
                            ok: 1,
                        }
                    }

                }
                catch (err) {

                    //已關閉(含操作進行中被close)影響後續所有筆數, 屬整批性錯誤(T4), 往外拋令整批reject, 不降為該筆ok為0
                    if (isClosed()) {
                        throw new Error(getClosedMsg('save'))
                    }

                    //交易回調內已設inserted, 惟交易失敗時未實際寫入, 須重設以免發出insert事件
                    inserted = false

                    //本筆失敗不中斷整批, 以ok為0並附err回報, 由呼叫端逐筆檢查
                    rest = {
                        n: 1,
                        nInserted: 0,
                        nModified: 0,
                        ok: 0,
                        err: getErrMsg(err),
                    }

                }

                //emit, 原autoInsert係轉呼叫insert故會發出insert事件, 此處維持既有行為,
                //須於交易外發出, 避免外部訂閱函數於寫交易內執行
                if (inserted) {

                    //update, 不能保證插入多少, 一律重設快取
                    _cache = null

                    //evs, 待整批定案後發出
                    evs.push(() => {
                        emitChange('insert', [v], rest)
                    })

                }

                //evs, 逐筆失敗之error事件, 待整批定案後發出, 每筆一次
                //此事件之發出不表示整批失敗, 整批仍resolve, 該筆以ok為0回報
                if (rest.ok === 0) {
                    evs.push(() => {
                        emitError('save', [v], rest.err)
                    })
                }

                return rest
            })

        }
        catch (err) {
            isErr = true
            res = err
        }

        //update, 不能保證變更多少, 一律重設快取
        _cache = null

        //emit, 於change可能須使用select, 故須放在重設快取之後
        //逐筆事件依序先發出, 故必早於本整批change
        if (!isErr) {
            each(evs, (fn) => {
                fn()
            })
            emitChange('save', data, res)
        }

        //check
        if (isErr) {

            //emit, 整批性錯誤須於reject之前發出
            emitError('save', data, res)

            return Promise.reject(res)
        }

        return res
    }

    /**
     * 刪除數據
     *
     * @memberOf WOrmLmdb
     * @param {Object|Array} data 輸入數據物件或陣列
     * @returns {Promise} 回傳Promise，resolve回傳與輸入等長之刪除結果陣列，各筆為{n,nDeleted,ok}，n為主鍵命中筆數(主鍵經LMDB鍵編碼後超過1978位元組者視為未命中)，單筆失敗或未給有效id者ok為0並附err，reject回傳錯誤訊息；處理中途發生整批性錯誤(如操作進行中被close)時整批reject，先前已刪除之筆不回滾且其逐筆事件不發出，呼叫端須重新讀取確認
     */
    async function del(data) {

        //check client, 已關閉(含關閉進行中)為終態, 於任何檢查與寫入之前先行攔下(T4: 實例已關閉屬整批性錯誤)
        if (isClosed()) {
            return procClosed('del', data)
        }
        let isErr = false

        //check
        if (!iseobj(data) && !isearr(data)) {
            return []
        }

        //res
        let res = null

        //evs, 逐筆之事件先緩存, 待整批定案(resolve)後依序發出; 整批性錯誤而reject者不發出(T10.1第1條)
        let evs = []

        try {

            //cloneDeep, 與外部數據脫勾; 置於try內, 無法複製之輸入同屬整批性錯誤而先發出error事件
            data = cloneDeep(data)

            //check
            if (!isarr(data)) {
                data = [data]
            }

            //pmSeries
            res = await pmSeries(data, async(v) => {

                //rest
                let rest = null

                try {

                    //id
                    let id = get(v, 'id', '')

                    //check
                    if (!isestr(id)) {
                        //未給有效v.id視為該筆數據有問題而無法處理, 以ok為0並附err回報,
                        //與[已給v.id但查無數據]之ok為1有別, 二者須可由ok分辨
                        //註: 此處不可直接return, 否則會跳過本回調末尾之error事件發出
                        rest = {
                            n: 0,
                            nDeleted: 0,
                            ok: 0,
                            err: `can not delete by invalid id[${id}]`,
                        }
                    }
                    else {

                        //查找資料表內v.id, 讀取之錯誤由下方catch處置
                        let vv = await getValue(id, 'del')

                        //check
                        if (iseobj(vv)) {
                            //已存在v.id則須刪除

                            //del
                            await client.del(id)

                            //rest
                            rest = {
                                n: 1,
                                nDeleted: 1,
                                ok: 1,
                            }

                        }
                        else {
                            //不存在v.id則不刪除, 未命中故n為0
                            rest = {
                                n: 0,
                                nDeleted: 0,
                                ok: 1,
                            }

                        }

                    }

                }
                catch (err) {

                    //已關閉(含操作進行中被close)影響後續所有筆數, 屬整批性錯誤(T4), 往外拋令整批reject, 不降為該筆ok為0
                    if (isClosed()) {
                        throw new Error(getClosedMsg('del'))
                    }

                    //本筆失敗不中斷整批, 以ok為0並附err回報, 由呼叫端逐筆檢查
                    rest = {
                        n: 1,
                        nDeleted: 0,
                        ok: 0,
                        err: getErrMsg(err),
                    }

                }

                //evs, 逐筆失敗之error事件, 待整批定案後發出, 每筆一次
                //此事件之發出不表示整批失敗, 整批仍resolve, 該筆以ok為0回報
                if (rest.ok === 0) {
                    evs.push(() => {
                        emitError('del', [v], rest.err)
                    })
                }

                return rest
            })

        }
        catch (err) {
            isErr = true
            res = err
        }

        //update, 不能保證刪除多少, 一律重設快取
        _cache = null

        //emit, 於change可能須使用select, 故須放在重設快取之後
        //逐筆事件依序先發出, 故必早於本整批change
        if (!isErr) {
            each(evs, (fn) => {
                fn()
            })
            emitChange('del', data, res)
        }

        //check
        if (isErr) {

            //emit, 整批性錯誤須於reject之前發出
            emitError('del', data, res)

            return Promise.reject(res)
        }

        return res
    }

    /**
     * 依條件刪除多筆數據(破壞性操作，須審慎使用，非必要不用；刪除單筆請用del)
     *
     * 使用須知:
     * 1. 不給條件即刪除全部數據: find未給、為undefined或為{}時，刪除資料表內全部數據，不另行確認。
     *    給了條件才是部分刪除。
     * 2. 傳了值卻不是條件物件(null、陣列含[]與[{...}]、字串、數值、布林、函數、Date、RegExp、Map、Set、類別實例等)
     *    一律整批reject，不視為{}，不刪除任何數據。
     * 3. 條件僅支援系列白名單: 欄位等值{欄:純量}(多欄為AND)、欄位運算子$eq $ne $gt $gte $lt $lte(值為純量)與
     *    $in $nin(值為純量陣列)、文件層$and $or(值為條件物件之非空陣列)。純量為字串(不含NUL字元與孤立代理字元)、有限數字、布林、有效Date；
     *    null僅可用於欄位等值、$eq、$ne。白名單外之運算子($not $nor $regex $options $exists $where $expr $like $ilike等)、
     *    欄位層$and/$or、子查詢、函數值、Symbol鍵、undefined值、空的$nin/$and/$or等一律整批reject，
     *    訊息為invalid find: 原因 at [路徑]，且不刪除任何數據。
     * 4. 與select共用同一檢核與同一條件，select(find)命中之數據即本函數刪除之數據；
     *    建議先以select(find)確認命中範圍再刪除(useCache開啟且他方已寫入時，select可能回傳快取，而本函數一律直讀資料庫)。
     * 5. 條件值若來自可能為undefined、null或空陣列之變數，呼叫端須先自行判斷；undefined值會reject而非被忽略，
     *    空的$nin會reject，$in:[]則為恆假(刪除0筆)。
     * 6. 語意差異(依後端，見README): MongoDB式: $ne(比較值非null時)與$nin會命中欄位為null或缺欄之數據，$ne:null則排除之；
     *    欄名同於Object.prototype成員(constructor、toString等)者，數據無此自有欄位時視為缺欄。
     *    需明確表達時可用可攜寫法: 排除NULL為{$and:[{f:{$ne:null}},{f:{$ne:x}}]}，含NULL為{$or:[{f:null},{f:{$ne:x}}]}。
     *
     * n與nDeleted皆為實際刪除筆數，不取全表筆數。條件無命中時回{n:0,nDeleted:0,ok:1}，不視為錯誤。
     * 處理中途發生整批性錯誤(如操作進行中被close)時整批reject，先前已刪除之筆不回滾，呼叫端須重新讀取確認；刪除依資料庫之實際鍵。
     *
     * @memberOf WOrmLmdb
     * @param {Object} [find={}] 輸入刪除條件物件，不給予find則代表刪除全部數據
     * @returns {Promise} 回傳Promise，resolve回傳刪除結果物件{n,nDeleted,ok}，reject回傳錯誤訊息
     */
    async function delAll(find = {}) {

        //check client, 已關閉(含關閉進行中)為終態, 於任何檢查與寫入之前先行攔下(T4: 實例已關閉屬整批性錯誤)
        if (isClosed()) {
            return procClosed('delAll', null)
        }

        //fc, 條件之入口檢核(R-FIND-01): 入口順序為closed→find檢核→讀取, 不合者依T10.3先發出error事件再reject而不讀取、不刪除、不重設快取,
        //訊息原樣(不經下方刪除階段之catch: 該處遇實例已關閉會改寫為closed訊息, 不得套用於檢核之錯誤);
        //其後一律只用檢核所回傳之全新複本(null原型, 與呼叫端之find脫勾), 與select所用者同一規則
        let fc = null
        try {
            fc = cvOrmFind(find)
        }
        catch (err) {
            emitError('delAll', null, err)
            return Promise.reject(err)
        }

        let isErr = false

        //res
        let res = null
        try {

            //ltkv, 一律直讀資料庫而不經快取, 刪除對象不可依可能過期之快取決定
            //刪除依資料庫之實際鍵而非數據之id欄位: 數據不帶有效id者(如他方寫入)以id刪除會拋錯, 致整表無從刪除
            let ltkv = readAll('delAll', true)

            //kps, 須刪除之鍵, 有條件者逐筆以同一查詢條件判定是否命中, 複本無自有鍵(未給、undefined、{})為全部
            //判定逐筆進行而不另建主鍵對照, 故無以一般物件加in判斷時之原型鏈誤判(主鍵為constructor、toString等)
            let kps = []
            if (Reflect.ownKeys(fc).length > 0) {
                let match = mkMatch(fc)
                for (let kv of ltkv) {
                    if (match(kv.value)) {
                        kps.push(kv.key)
                    }
                }
            }
            else {
                kps = map(ltkv, 'key')
            }
            let nDel = size(kps)

            //del
            for (let k of kps) {
                await client.del(k)
            }

            //res, n為實際刪除筆數而非全表筆數, 故與nDeleted同值
            res = {
                n: nDel,
                nDeleted: nDel,
                ok: 1,
            }

        }
        catch (err) {
            isErr = true
            res = err

            //刪除途中被close者, lmdb回報之訊息不一, 統一為已關閉之訊息(T4: 訊息明示closed)
            if (isClosed()) {
                res = new Error(getClosedMsg('delAll'))
            }

        }

        //update, 不能保證刪除多少, 一律重設快取
        _cache = null

        //emit, 於change可能須使用select, 故須放在重設快取之後
        if (!isErr) {
            emitChange('delAll', null, res)
        }

        //check
        if (isErr) {

            //emit, 整批性錯誤須於reject之前發出
            emitError('delAll', null, res)

            return Promise.reject(res)
        }

        return res
    }

    //close, 關閉內部LMDB env並釋放檔案鎖, close後該實例即報廢, 須重新new WOrmLmdb,
    //呼叫close()之後(含其未完成之關閉進行中)再呼叫任何操作函數將立即reject(訊息明示closed), 不與[查無資料]同形;
    //操作進行中被close者, 其後續步驟亦以整批性錯誤reject; 重複呼叫close()無妨, lmdb於關閉中與已關閉時皆正常回傳
    let close = async () => {
        if (client && client.status !== 'closed') {
            await client.close() //lmdb root database之close(), 回傳Promise
        }
    }

    //save
    ee.select = select
    ee.selectByPk = selectByPk
    ee.insert = insert
    ee.insertBulk = insertBulk
    ee.save = save
    ee.del = del
    ee.delAll = delAll
    ee.close = close

    return ee
}
